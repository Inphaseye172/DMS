cmake_minimum_required(VERSION 3.25)
project(DMS_Depth VERSION 1.0.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

option(DMS_BUILD_PLUGIN   "Build the After Effects plugin (.aex); requires AE_SDK_ROOT (Windows only)" ${WIN32})
option(DMS_BUILD_CLI      "Build the depthcli tool" ON)
option(DMS_BUILD_TESTS    "Build unit tests (GoogleTest)" ON)
option(DMS_ENABLE_ASAN    "Enable AddressSanitizer/UBSan for engine, CLI and tests (Debug)" OFF)
option(DMS_ENABLE_TENSORRT "Compile the optional TensorRT EP code path (CUDA builds only)" OFF)
option(DMS_ENABLE_CLANG_TIDY "Run clang-tidy during the build" OFF)
option(DMS_ENABLE_MSVC_ANALYZE "Run MSVC /analyze during the build" OFF)
option(DMS_FETCH_ORT      "Download the pinned ONNX Runtime package at configure time" ON)

list(APPEND CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}/cmake")
include(Warnings)
include(Analyze)
include(FetchORT)

add_subdirectory(third_party)
add_subdirectory(engine)
add_subdirectory(plugin/core)
if(DMS_BUILD_CLI)
  add_subdirectory(tools/depthcli)
endif()
if(DMS_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests/unit)
endif()
if(DMS_BUILD_PLUGIN)
  if(NOT WIN32)
    message(FATAL_ERROR "DMS_BUILD_PLUGIN=ON requires Windows + MSVC (After Effects plugins are .aex DLLs). Configure with -DDMS_BUILD_PLUGIN=OFF to build the engine/CLI/tests only.")
  endif()
  add_subdirectory(plugin)
endif()

# Compile-only check of the AE glue against tools/ae_sdk_mock (Linux/CI). Not a plugin build.
option(DMS_AE_GLUE_CHECK "Compile plugin/ae against the AE SDK mock (no linking) when the real plugin is not built" ON)
if(DMS_AE_GLUE_CHECK AND NOT DMS_BUILD_PLUGIN)
  add_library(dms_ae_glue_check OBJECT
    plugin/ae/DMS_Depth.cpp plugin/ae/DMS_Depth_Params.cpp plugin/ae/DMS_Depth_Render.cpp
    plugin/ae/DMS_Depth_Dialog.cpp plugin/ae/DMS_Depth_GPU.cpp)
  target_include_directories(dms_ae_glue_check PRIVATE plugin/ae)
  target_include_directories(dms_ae_glue_check SYSTEM PRIVATE tools/ae_sdk_mock)
  target_link_libraries(dms_ae_glue_check PRIVATE dms_plugin_core dms_engine)
  dms_apply_warnings(dms_ae_glue_check)
endif()
